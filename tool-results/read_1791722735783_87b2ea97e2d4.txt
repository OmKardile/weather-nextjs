    99→/* ------------------------------------------------------------------ */
   100→/* Visibility gating — ONE IntersectionObserver shared by every icon.  */
   101→/* Leaving the viewport adds `.icon-paused` to the root <svg>.         */
   102→/* Ancient browsers without IntersectionObserver simply never pause.  */
   103→/* ------------------------------------------------------------------ */
   104→
   105→let iconVisibilityObserver: IntersectionObserver | null | undefined;
   106→
   107→function getIconVisibilityObserver(): IntersectionObserver | null {
   108→  if (iconVisibilityObserver !== undefined) return iconVisibilityObserver;
   109→  if (typeof window === 'undefined' || typeof IntersectionObserver === 'undefined') {
   110→    iconVisibilityObserver = null;
   111→    return null;
   112→  }
   113→  iconVisibilityObserver = new IntersectionObserver(
   114→    (entries) => {
   115→      for (const entry of entries) {
   116→        entry.target.classList.toggle('icon-paused', !entry.isIntersecting);
   117→      }
   118→    },
   119→    { rootMargin: '32px 0px' },
   120→  );
   121→  return iconVisibilityObserver;
   122→}

   226→/** Candy sun — radial-lit core, rotating rays, warm aura. */
   227→function SunGlyph({
   228→  id,
   229→  cx,
   230→  cy,
   231→  r,
   232→  rays = true,
   233→}: {
   234→  id: string;
   235→  cx: number;
   236→  cy: number;
   237→  r: number;
   238→  rays?: boolean;
   239→}) {
   240→  const rIn = r * 1.5;
   241→  const rOut = r * 2.1;
   242→  return (
   243→    <>
   244→      <defs>
   245→        <radialGradient id={`${id}-sc`} cx="0.36" cy="0.32" r="0.85">
   246→          <stop offset="0" stopColor={PAL.sunCore[0]} />
   247→          <stop offset="0.42" stopColor={PAL.sunCore[1]} />
   248→          <stop offset="1" stopColor={PAL.sunCore[2]} />
   249→        </radialGradient>
   250→        <radialGradient id={`${id}-sg`} cx="0.5" cy="0.5" r="0.5">
   251→          <stop offset="0" stopColor="#F0954A" stopOpacity="0.22" />
   252→          <stop offset="0.72" stopColor="#F0954A" stopOpacity="0.08" />
   253→          <stop offset="1" stopColor="#F0954A" stopOpacity="0" />
   254→        </radialGradient>
   255→        <radialGradient id={`${id}-sh`} cx="0.5" cy="0.38" r="0.62">
   256→          <stop offset="0" stopColor="#ffffff" stopOpacity="0.65" />
   257→          <stop offset="1" stopColor="#ffffff" stopOpacity="0" />
   258→        </radialGradient>
   259→      </defs>
   260→      {/* warm aura */}
   261→      <circle cx={cx} cy={cy} r={r * 2.5} fill={`url(#${id}-sg)`} />
   262→      {rays && (
   263→        <g className="ic-sun-rays" stroke={PAL.ray} strokeWidth={r * 0.3} strokeLinecap="round">
   264→          {RAY_ANGLES.map((deg) => {
   265→            const rad = (deg * Math.PI) / 180;
   266→            const cos = Math.cos(rad);
   267→            const sin = Math.sin(rad);
   268→            return (
   269→              <line
   270→                key={deg}
   271→                x1={cx + cos * rIn}
   272→                y1={cy + sin * rIn}
   273→                x2={cx + cos * rOut}
   274→                y2={cy + sin * rOut}
   275→              />
   276→            );
   277→          })}
   278→        </g>
   279→      )}
   280→      <g className="ic-sun-core">
   281→        <circle cx={cx} cy={cy} r={r} fill={`url(#${id}-sc)`} />
   282→        <ellipse
   283→          cx={cx - r * 0.3}
   284→          cy={cy - r * 0.34}
   285→          rx={r * 0.42}
   286→          ry={r * 0.3}
   287→          fill={`url(#${id}-sh)`}
   288→          transform={`rotate(-18 ${cx - r * 0.3} ${cy - r * 0.34})`}
   289→        />
   290→      </g>
   291→    </>
   292→  );
   293→}

   881→
   882→export function Icon({ name, size = 24, animated = true, className, strokeWidth = 2 }: IconProps) {
   883→  // useId → collision-free ids for gradients / masks / clipPaths
   884→  // (10+ icons share the hourly strip page).
   885→  const rawId = useId();
   886→  const id = `a${rawId.replace(/[^a-zA-Z0-9]/g, '')}`;
   887→  const rootRef = useRef<SVGSVGElement | null>(null);
   888→  const isWeather = WEATHER_NAMES.has(name);
   889→
   890→  // Pause looping weather animations while off-screen (one shared observer).
   891→  useEffect(() => {
   892→    if (!isWeather || !animated) return;
   893→    const el = rootRef.current;
   894→    if (!el) return;
   895→    const observer = getIconVisibilityObserver();
   896→    if (!observer) return;
   897→    observer.observe(el);
   898→    return () => {
   899→      observer.unobserve(el);
   900→      el.classList.remove('icon-paused');
   901→    };
   902→  }, [isWeather, animated]);
   903→
   904→  return (
   905→    <svg
   906→      ref={rootRef}
   907→      viewBox={isWeather ? '0 0 64 64' : '0 0 24 24'}
   908→      width={size}
   909→      height={size}
   910→      aria-hidden="true"
   911→      focusable="false"
   912→      className={cn('shrink-0', animated && 'ic-anim', !isWeather && 'ui-icon', className)}
   913→    >
   914→      {isWeather
   915→        ? WEATHER_GLYPHS[name as WeatherIconName](id)
   916→        : UI_GLYPHS[name as UiIconName]({ id, sw: strokeWidth })}
   917→    </svg>
   918→  );
   919→}
