    57→class UpstreamError extends Error {
    58→  constructor(
    59→    message: string,
    60→    readonly status = 502,
    61→  ) {
    62→    super(message);
    63→    this.name = 'UpstreamError';
    64→  }

    69→/** Upstream attempts before giving up (the sandbox egress path needs a few tries for cold hosts). */
    70→const FETCH_ATTEMPTS = 5;
    71→
    72→async function fetchJson<T>(url: string, revalidate: number, init?: RequestInit): Promise<T> {
    73→  // Transient network hiccups (this host's upstream route occasionally times
    74→  // out, and first contact with an upstream host can ETIMEDOUT a few times
    75→  // before the egress path warms up) get up to 5 attempts with escalating
    76→  // backoff (250ms → 1s → 2.25s → 4s).
    77→  let lastError: unknown;
    78→  for (let attempt = 1; attempt <= FETCH_ATTEMPTS; attempt += 1) {
    79→    try {
    80→      const res = await fetch(url, { ...init, next: { revalidate } });
    81→      if (!res.ok) {
    82→        throw new UpstreamError(
    83→          `Upstream request failed (${res.status}) for ${new URL(url).hostname}`,
    84→        );
    85→      }
    86→      return (await res.json()) as T;
    87→    } catch (error) {
    88→      lastError = error;
    89→      if (error instanceof UpstreamError) throw error;
    90→      // A caller-supplied deadline (the historical 15s cap) is final.
    91→      if (init?.signal?.aborted) throw error;
    92→      if (attempt < FETCH_ATTEMPTS) await sleep(250 * attempt * attempt);
    93→    }
    94→  }
    95→  throw lastError instanceof Error
    96→    ? lastError
    97→    : new UpstreamError('Upstream request failed');
    98→}

   304→const num = (value: number | null | undefined, fallback: number): number =>
   305→  value == null || !Number.isFinite(value) ? fallback : value;

   307→/** Slice the next 48 hours starting at (or just after) the current time. */
   308→function sliceCurrentHour(forecast: OmForecast): HourlyPoint[] {
   309→  const { hourly, current } = forecast;
   310→  const currentHourPrefix = current.time.slice(0, 13); // "YYYY-MM-DDTHH"
   311→  let start = hourly.time.findIndex((t) => t.slice(0, 13) === currentHourPrefix);
   312→  if (start < 0) start = Math.max(0, hourly.time.length - 48);
   313→
   314→  const points: HourlyPoint[] = [];
   315→  for (let i = start; i < hourly.time.length && points.length < 48; i += 1) {
   316→    points.push({
   317→      time: hourly.time[i],
   318→      temperature: num(hourly.temperature_2m[i], 0),
   319→      precipitationProbability: num(hourly.precipitation_probability[i], 0),
   320→      precipitation: num(hourly.precipitation[i], 0),
   321→      weatherCode: num(hourly.weather_code[i], 0),
   322→      isDay: num(hourly.is_day[i], 1) === 1,
   323→      visibility: num(hourly.visibility[i], 24_000),
   324→      uvIndex: num(hourly.uv_index[i], 0),
   325→      windSpeed: num(hourly.wind_speed_10m[i], 0),
   326→      condition: conditionOf(num(hourly.weather_code[i], 0)),
   327→    });
   328→  }
   329→  return points;
   330→}

   332→/**
   333→ * Slice the 15-minute nowcast from the current time slot → +2 hours.
   334→ * Returns null when the response is missing/empty (the call is best-effort).
   335→ */
   336→function sliceMinutely(data: OmMinutelyForecast | null, nowLocal: string): MinutelyPoint[] | null {
   337→  const minutely = data?.minutely_15;
   338→  if (!minutely || minutely.time.length === 0) return null;
   339→  // Local ISO strings are zero-padded, so lexicographic order === chronological order.
   340→  let start = minutely.time.findIndex((t) => t >= nowLocal);
   341→  if (start < 0) return null;
   342→
   343→  const points: MinutelyPoint[] = [];
   344→  for (let i = start; i < minutely.time.length && points.length < MINUTELY_POINTS; i += 1) {
   345→    points.push({
   346→      time: minutely.time[i],
   347→      precip: num(minutely.precipitation?.[i] ?? null, 0),
   348→      precipProbability: num(minutely.precipitation_probability?.[i] ?? null, 0),
   349→    });
   350→  }
   351→  return points.length > 0 ? points : null;
   352→}

   354→const POLLEN_KEYS = [
   355→  'alder_pollen',
   356→  'birch_pollen',
   357→  'grass_pollen',
   358→  'ragweed_pollen',
   359→  'olive_pollen',
   360→  'mugwort_pollen',
   361→] as const;
   362→
   363→/**
   364→ * Summarize hourly pollen into a single next-24h snapshot (max per type).
   365→ * `available` is true only when the upstream actually reports pollen here
   366→ * (Europe); elsewhere every array is all-null and the flag stays false.
   367→ */
   368→function extractPollen(air: OmAirQuality | null): Pollen | null {
   369→  const hourly = air?.hourly;
   370→  if (!hourly || hourly.time.length === 0) return null;
   371→
   372→  const anchor = air.current?.time ?? hourly.time[0];
   373→  let start = hourly.time.findIndex((t) => t >= anchor);
   374→  if (start < 0) start = 0;
   375→  const end = Math.min(start + 24, hourly.time.length);
   376→
   377→  const maxOf = (key: (typeof POLLEN_KEYS)[number]): number | null => {
   378→    let max: number | null = null;
   379→    const values = hourly?.[key];
   380→    if (!values) return null;
   381→    for (let i = start; i < end; i += 1) {
   382→      const value = values[i];
   383→      if (value != null && Number.isFinite(value) && (max == null || value > max)) {
   384→        max = value;
   385→      }
   386→    }
   387→    return max;
   388→  };
   389→
   390→  const [alder, birch, grass, ragweed, olive, mugwort] = POLLEN_KEYS.map(maxOf);
   391→  return {
   392→    available: [alder, birch, grass, ragweed, olive, mugwort].some((v) => v !== null),
   393→    time: hourly.time[start],
   394→    alder,
   395→    birch,
   396→    grass,
   397→    ragweed,
   398→    olive,
   399→    mugwort,
   400→  };
   401→}

   403→export async function fetchWeatherBundle(lat: number, lon: number): Promise<WeatherBundle> {
   404→  const forecastUrl =
   405→    `${FORECAST_URL}?latitude=${lat}&longitude=${lon}` +
   406→    `&current=${CURRENT_FIELDS}&hourly=${HOURLY_FIELDS}&daily=${DAILY_FIELDS}` +
   407→    `&timezone=auto&forecast_days=16&past_hours=1&forecast_hours=54`;
   408→
   409→  // Pollen piggybacks on the air-quality call (same upstream, one request).
   410→  const airUrl =
   411→    `${AIR_QUALITY_URL}?latitude=${lat}&longitude=${lon}` +
   412→    `&current=european_aqi,us_aqi,pm10,pm2_5,carbon_monoxide,nitrogen_dioxide,sulphur_dioxide,ozone` +
   413→    `&hourly=${POLLEN_FIELDS}&forecast_days=3&timezone=auto`;
   414→
   415→  // Dedicated nowcast call. forecast_days=2 (not 1) so the 2-hour window is
   416→  // fully covered even shortly before local midnight.
   417→  const minutelyUrl =
   418→    `${FORECAST_URL}?latitude=${lat}&longitude=${lon}` +
   419→    `&minutely_15=${MINUTELY_FIELDS}&forecast_days=2&timezone=auto`;
   420→
   421→  // Air quality, pollen and the nowcast are nice-to-haves: a failure of any
   422→  // of them must not break the core forecast.
   423→  const [forecast, air, minutelyForecast] = await Promise.all([
   424→    fetchJson<OmForecast>(forecastUrl, WEATHER_REVALIDATE),
   425→    fetchJson<OmAirQuality>(airUrl, WEATHER_REVALIDATE).catch(() => null),
   426→    fetchJson<OmMinutelyForecast>(minutelyUrl, WEATHER_REVALIDATE).catch(() => null),
   427→  ]);
   428→
   429→  const current = forecast.current;
   430→  const daily: DailyPoint[] = forecast.daily.time.slice(0, 16).map((date, i) => ({
   431→    date,
   432→    weatherCode: num(forecast.daily.weather_code[i], 0),
   433→    tempMax: num(forecast.daily.temperature_2m_max[i], 0),
   434→    tempMin: num(forecast.daily.temperature_2m_min[i], 0),
   435→    sunrise: forecast.daily.sunrise[i] ?? `${date}T06:00`,
   436→    sunset: forecast.daily.sunset[i] ?? `${date}T18:00`,
   437→    precipitationProbabilityMax: num(forecast.daily.precipitation_probability_max[i], 0),
   438→    uvIndexMax: num(forecast.daily.uv_index_max[i], 0),
   439→    windSpeedMax: num(forecast.daily.wind_speed_10m_max[i], 0),
   440→    condition: conditionOf(num(forecast.daily.weather_code[i], 0)),
   441→  }));
   442→
   443→  const airQuality: AirQuality | null = air?.current
   444→    ? {
   445→        europeanAqi: air.current.european_aqi ?? null,
   446→        usAqi: air.current.us_aqi ?? null,
   447→        pm2_5: air.current.pm2_5 ?? null,
   448→        pm10: air.current.pm10 ?? null,
   449→        ozone: air.current.ozone ?? null,
   450→        no2: air.current.nitrogen_dioxide ?? null,
   451→        so2: air.current.sulphur_dioxide ?? null,
   452→        co: air.current.carbon_monoxide ?? null,
   453→        time: air.current.time,
   454→      }
   455→    : null;
   456→
   457→  return {
   458→    current: {
   459→      time: current.time,
   460→      temperature: num(current.temperature_2m, 0),
   461→      apparentTemperature: num(current.apparent_temperature, 0),
   462→      humidity: num(current.relative_humidity_2m, 0),
   463→      isDay: num(current.is_day, 1) === 1,
   464→      precipitation: num(current.precipitation, 0),
   465→      weatherCode: num(current.weather_code, 0),
   466→      cloudCover: num(current.cloud_cover, 0),
   467→      pressure: num(current.pressure_msl, 1013),
   468→      windSpeed: num(current.wind_speed_10m, 0),
   469→      windGust: num(current.wind_gusts_10m, 0),
   470→      windDirection: num(current.wind_direction_10m, 0),
   471→      condition: conditionOf(num(current.weather_code, 0)),
   472→    },
   473→    hourly: sliceCurrentHour(forecast),
   474→    daily,
   475→    airQuality,
   476→    minutely: sliceMinutely(minutelyForecast, current.time),
   477→    pollen: extractPollen(air),
   478→    utcOffsetSeconds: forecast.utc_offset_seconds ?? 0,
   479→    timezone: forecast.timezone ?? 'UTC',
   480→    timezoneAbbreviation: forecast.timezone_abbreviation ?? '',
   481→    fetchedAt: new Date().toISOString(),
   482→  };
   483→}

   523→/** Side-by-side ECMWF / GFS / ICON 7-day hourly comparison. */
   524→export async function fetchModelComparison(lat: number, lon: number): Promise<ModelsResponse> {
   525→  const url =
   526→    `${FORECAST_URL}?latitude=${lat}&longitude=${lon}` +
   527→    `&models=${MODELS_PARAM}&hourly=temperature_2m,precipitation&forecast_days=7&timezone=auto`;

   533→    label,
   534→    hourly: (hourly?.time ?? []).map<ModelHourlyPoint>((time, i) => ({
   535→      time,
   536→      temp: num(hourly?.[tempKey]?.[i] ?? null, 0),
   537→      precip: num(hourly?.[precipKey]?.[i] ?? null, 0),
   538→    })),
   539→  }));
   540→  return { models };
   541→}

   543→/** ICON ensemble: base series + member01…member39. */
   544→const ENSEMBLE_MEMBERS = 40;
   545→
   546→interface OmEnsembleHourly {
   547→  time: string[];
   548→  [variable: string]: unknown;
   549→}
   550→
   551→function numSeries(hourly: OmEnsembleHourly, key: string): Array<number | null> {
   552→  const raw = hourly[key];
   553→  if (!Array.isArray(raw)) return [];
   554→  const values: Array<number | null> = [];
   555→  for (const item of raw) {
   556→    if (item === null || typeof item === 'number') values.push(item);
   557→  }
   558→  return values;
   559→}
   560→
   561→/**
   562→ * ICON ensemble mean ± spread for the next 7 days. The upstream `*_spread`
   563→ * arrays currently come back all-null, so the spread is computed here as the
   564→ * population σ across the 40 returned member series (upstream value wins if
   565→ * it ever starts reporting).
   566→ */
   567→export async function fetchEnsemble(lat: number, lon: number): Promise<EnsembleResponse> {
   568→  const url =
   569→    `${ENSEMBLE_URL}?latitude=${lat}&longitude=${lon}` +
   570→    `&models=icon_seamless&hourly=temperature_2m,temperature_2m_spread,precipitation` +
   571→    `&forecast_days=7&timezone=auto`;
   572→  const data = await fetchJson<{ hourly?: OmEnsembleHourly }>(url, MODELS_REVALIDATE);
   573→  const hourly = data.hourly;
   574→  if (!hourly) return { hourly: [] };
   575→
   576→  const temps = numSeries(hourly, 'temperature_2m');
   577→  const precip = numSeries(hourly, 'precipitation');
   578→  const upstreamSpread = numSeries(hourly, 'temperature_2m_spread');
   579→  const memberTemps = [
   580→    temps,
   581→    ...Array.from({ length: ENSEMBLE_MEMBERS - 1 }, (_, m) =>
   582→      numSeries(hourly, `temperature_2m_member${String(m + 1).padStart(2, '0')}`),
   583→    ),
   584→  ].filter((series) => series.length > 0);
   585→
   586→  // Trim trailing hours the ensemble horizon doesn't cover (ICON ~142 of 168).
   587→  let last = hourly.time.length - 1;
   588→  while (last >= 0 && temps[last] == null && precip[last] == null) last -= 1;
   589→
   590→  const points: EnsembleHourly[] = [];
   591→  for (let i = 0; i <= last; i += 1) {
   592→    let tempSpread = upstreamSpread[i] ?? null;
   593→    if (tempSpread == null && memberTemps.length > 1) {
   594→      const values: number[] = [];
   595→      for (const series of memberTemps) {
   596→        const value = series[i];
   597→        if (value != null) values.push(value);
   598→      }
   599→      if (values.length > 1) {
   600→        const mean = values.reduce((a, v) => a + v, 0) / values.length;
   601→        const variance = values.reduce((a, v) => a + (v - mean) ** 2, 0) / values.length;
   602→        tempSpread = Math.round(Math.sqrt(variance) * 100) / 100;
   603→      }
   604→    }
   605→    points.push({
   606→      time: hourly.time[i],
   607→      temp: num(temps[i], 0),
   608→      tempSpread,
   609→      precip: num(precip[i], 0),
   610→    });
   611→  }
   612→  return { hourly: points };
   613→}

   619→const HISTORICAL_START_YEAR = 1995;
   620→
   621→interface OmArchiveDaily {
   622→  time: string[];
   623→  temperature_2m_max?: (number | null)[];
   624→  temperature_2m_min?: (number | null)[];

   627→/**
   628→ * 1995 → last complete year ERA5 history for one calendar date, plus the
   629→ * 30-year normal. One upstream call per rounded coordinate (the archive
   630→ * payload is shared across every month/day of that spot via the fetch cache).
   631→ * Feb 29 keeps its year slot with nulls in non-leap years so the year axis
   632→ * stays continuous.
   633→ */
   634→export async function fetchHistorical(
   635→  lat: number,
   636→  lon: number,
   637→  month: number,
   638→  day: number,
   639→): Promise<HistoricalResponse> {
   640→  // Round to 1 decimal so nearby queries share a single cached request.
   641→  const rlat = Math.round(lat * 10) / 10;
   642→  const rlon = Math.round(lon * 10) / 10;
   643→  const endYear = new Date().getUTCFullYear() - 1;
   644→  const url =
   645→    `${ARCHIVE_URL}?latitude=${rlat}&longitude=${rlon}` +
   646→    `&start_date=${HISTORICAL_START_YEAR}-01-01&end_date=${endYear}-12-31` +
   647→    `&daily=temperature_2m_max,temperature_2m_min&timezone=auto`;
   648→
   649→  // Hard 15s cap: a slow archive fetch surfaces as an error at the route
   650→  // level (which degrades to a 200 { error } response). The aborted signal
   651→  // also short-circuits fetchJson's retries, keeping the total ≈ 15s.
   652→  const data = await fetchJson<{ daily?: OmArchiveDaily }>(url, HISTORICAL_REVALIDATE, {
   653→    signal: AbortSignal.timeout(HISTORICAL_TIMEOUT_MS),
   654→  });
   655→
   656→  const times = data.daily?.time ?? [];
   657→  const tmaxSeries = data.daily?.temperature_2m_max ?? [];
   658→  const tminSeries = data.daily?.temperature_2m_min ?? [];
   659→
   660→  const suffix = `-${String(month).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
   661→  const byYear = new Map<number, { tmax: number | null; tmin: number | null }>();
   662→  for (let i = 0; i < times.length; i += 1) {
   663→    const date = times[i];
   664→    if (!date.endsWith(suffix)) continue;
   665→    byYear.set(Number(date.slice(0, 4)), {
   666→      tmax: tmaxSeries[i] ?? null,
   667→      tmin: tminSeries[i] ?? null,
   668→    });
   669→  }
   670→
   671→  const firstYear = times.length > 0 ? Number(times[0].slice(0, 4)) : HISTORICAL_START_YEAR;
   672→  const lastYear = times.length > 0 ? Number(times[times.length - 1].slice(0, 4)) : endYear;
   673→
   674→  const years: HistoricalYear[] = [];
   675→  for (let year = firstYear; year <= lastYear; year += 1) {
   676→    const hit = byYear.get(year);
   677→    years.push({ year, tmax: hit?.tmax ?? null, tmin: hit?.tmin ?? null });
   678→  }
   679→
   680→  const mean = (values: number[]): number | null =>
   681→    values.length === 0
   682→      ? null
   683→      : Math.round((values.reduce((a, v) => a + v, 0) / values.length) * 10) / 10;
   684→
   685→  return {
   686→    month,
   687→    day,
   688→    years,
   689→    normal: {
   690→      tmax: mean(years.map((y) => y.tmax).filter((v): v is number => v != null)),
   691→      tmin: mean(years.map((y) => y.tmin).filter((v): v is number => v != null)),
   692→    },
   693→  };
   694→}
