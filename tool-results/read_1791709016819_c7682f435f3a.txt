    93→/* ------------------------------------------------------------------ */
    94→/* Visibility gating — ONE IntersectionObserver shared by every icon.  */
    95→/* Leaving the viewport adds `.icon-paused` to the root <svg>.         */
    96→/* Ancient browsers without IntersectionObserver simply never pause.  */
    97→/* ------------------------------------------------------------------ */
    98→
    99→let iconVisibilityObserver: IntersectionObserver | null | undefined;
   100→
   101→function getIconVisibilityObserver(): IntersectionObserver | null {
   102→  if (iconVisibilityObserver !== undefined) return iconVisibilityObserver;
   103→  if (typeof window === 'undefined' || typeof IntersectionObserver === 'undefined') {
   104→    iconVisibilityObserver = null;
   105→    return null;
   106→  }
   107→  iconVisibilityObserver = new IntersectionObserver(
   108→    (entries) => {
   109→      for (const entry of entries) {
   110→        entry.target.classList.toggle('icon-paused', !entry.isIntersecting);
   111→      }
   112→    },
   113→    { rootMargin: '32px 0px' },
   114→  );
   115→  return iconVisibilityObserver;
   116→}

   220→/** Candy sun — radial-lit core, rotating rays, warm aura. */
   221→function SunGlyph({
   222→  id,
   223→  cx,
   224→  cy,
   225→  r,
   226→  rays = true,
   227→}: {
   228→  id: string;
   229→  cx: number;
   230→  cy: number;
   231→  r: number;
   232→  rays?: boolean;
   233→}) {
   234→  const rIn = r * 1.5;
   235→  const rOut = r * 2.1;
   236→  return (
   237→    <>
   238→      <defs>
   239→        <radialGradient id={`${id}-sc`} cx="0.36" cy="0.32" r="0.85">
   240→          <stop offset="0" stopColor={PAL.sunCore[0]} />
   241→          <stop offset="0.42" stopColor={PAL.sunCore[1]} />
   242→          <stop offset="1" stopColor={PAL.sunCore[2]} />
   243→        </radialGradient>
   244→        <radialGradient id={`${id}-sg`} cx="0.5" cy="0.5" r="0.5">
   245→          <stop offset="0" stopColor="#F0954A" stopOpacity="0.22" />
   246→          <stop offset="0.72" stopColor="#F0954A" stopOpacity="0.08" />
   247→          <stop offset="1" stopColor="#F0954A" stopOpacity="0" />
   248→        </radialGradient>
   249→        <radialGradient id={`${id}-sh`} cx="0.5" cy="0.38" r="0.62">
   250→          <stop offset="0" stopColor="#ffffff" stopOpacity="0.65" />
   251→          <stop offset="1" stopColor="#ffffff" stopOpacity="0" />
   252→        </radialGradient>
   253→      </defs>
   254→      {/* warm aura */}
   255→      <circle cx={cx} cy={cy} r={r * 2.5} fill={`url(#${id}-sg)`} />
   256→      {rays && (
   257→        <g className="ic-sun-rays" stroke={PAL.ray} strokeWidth={r * 0.3} strokeLinecap="round">
   258→          {RAY_ANGLES.map((deg) => {
   259→            const rad = (deg * Math.PI) / 180;
   260→            const cos = Math.cos(rad);
   261→            const sin = Math.sin(rad);
   262→            return (
   263→              <line
   264→                key={deg}
   265→                x1={cx + cos * rIn}
   266→                y1={cy + sin * rIn}
   267→                x2={cx + cos * rOut}
   268→                y2={cy + sin * rOut}
   269→              />
   270→            );
   271→          })}
   272→        </g>
   273→      )}
   274→      <g className="ic-sun-core">
   275→        <circle cx={cx} cy={cy} r={r} fill={`url(#${id}-sc)`} />
   276→        <ellipse
   277→          cx={cx - r * 0.3}
   278→          cy={cy - r * 0.34}
   279→          rx={r * 0.42}
   280→          ry={r * 0.3}
   281→          fill={`url(#${id}-sh)`}
   282→          transform={`rotate(-18 ${cx - r * 0.3} ${cy - r * 0.34})`}
   283→        />
   284→      </g>
   285→    </>
   286→  );
   287→}

   832→export function Icon({ name, size = 24, animated = true, className, strokeWidth = 2 }: IconProps) {
   833→  // useId → collision-free ids for gradients / masks / clipPaths
   834→  // (10+ icons share the hourly strip page).
   835→  const rawId = useId();
   836→  const id = `a${rawId.replace(/[^a-zA-Z0-9]/g, '')}`;
   837→  const rootRef = useRef<SVGSVGElement | null>(null);
   838→  const isWeather = WEATHER_NAMES.has(name);
   839→
   840→  // Pause looping weather animations while off-screen (one shared observer).
   841→  useEffect(() => {
   842→    if (!isWeather || !animated) return;
   843→    const el = rootRef.current;
   844→    if (!el) return;
   845→    const observer = getIconVisibilityObserver();
   846→    if (!observer) return;
   847→    observer.observe(el);
   848→    return () => {
   849→      observer.unobserve(el);
   850→      el.classList.remove('icon-paused');
   851→    };
   852→  }, [isWeather, animated]);
   853→
   854→  return (
   855→    <svg
   856→      ref={rootRef}
   857→      viewBox={isWeather ? '0 0 64 64' : '0 0 24 24'}
   858→      width={size}
   859→      height={size}
   860→      aria-hidden="true"
   861→      focusable="false"
   862→      className={cn('shrink-0', animated && 'ic-anim', !isWeather && 'ui-icon', className)}
   863→    >
   864→      {isWeather
   865→        ? WEATHER_GLYPHS[name as WeatherIconName](id)
   866→        : UI_GLYPHS[name as UiIconName]({ id, sw: strokeWidth })}
   867→    </svg>
   868→  );
   869→}
