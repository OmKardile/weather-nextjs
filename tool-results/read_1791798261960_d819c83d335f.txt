    76→function normalizeQuake(p: Record<string, unknown>): GeoQuake {
    77→  return {
    78→    id: String(p.id ?? ''),
    79→    magnitude: Number(p.magnitude ?? 0),
    80→    place: String(p.place ?? ''),
    81→    time: String(p.time ?? new Date().toISOString()),
    82→    latitude: Number(p.latitude ?? 0),
    83→    longitude: Number(p.longitude ?? 0),
    84→    depthKm: Number(p.depthKm ?? 0),
    85→    url: String(p.url ?? ''),
    86→    tsunami: Boolean(p.tsunami),
    87→  };
    88→}
    89→
    90→function quakesToCollection(quakes: GeoQuake[]): FeatureCollection {
    91→  return {
    92→    type: 'FeatureCollection',
    93→    features: quakes.map((q) => ({
    94→      type: 'Feature' as const,
    95→      geometry: { type: 'Point' as const, coordinates: [q.longitude, q.latitude] },
    96→      properties: { ...q },
    97→    })),
    98→  };
    99→}

   161→  }, []);
   162→

   164→

   233→      minZoom: 1,
   234→      maxZoom: 17,
   235→      dragRotate: false,
   236→      touchPitch: false,
   237→      attributionControl: { compact: true },
   238→    });

   240→    instance.addControl(new maplibregl.NavigationControl({ showCompass: false }), 'top-left');
   241→    // Radar/satellite tiles 404 at the edges — expected, stay quiet.
   242→    instance.on('error', () => {});
   243→
   244→    const round1 = (v: number) => Math.round(v * 10) / 10;
   245→    const syncBbox = () => {
   246→      const b = instance.getBounds();
   247→      const next: ViewBbox = {
   248→        minLat: round1(b.getSouth()),
   249→        minLon: round1(b.getWest()),

   258→        prev.maxLat !== next.maxLat ||

   273→      setMap(instance);
   274→      syncBbox();

   277→    instance.on('moveend', onMoveEnd);

   297→      openPointPopup(e.lngLat.lng, e.lngLat.lat);
   298→    };
   299→    instance.on('click', onClick);
   300→
   301→    const onMouseMove = (e: maplibregl.MapMouseEvent) => {

   306→    instance.on('mousemove', onMouseMove);
   307→
   308→    return () => {
   309→      window.clearTimeout(bboxDebounce);
   310→      popupRef.current?.remove();
   311→      markerRef.current?.remove();
   312→      popupRef.current = null;
   313→      markerRef.current = null;
   314→      // Explicit removal of every dynamic layer/source, then the map itself.

   316→        if (instance.getLayer(id)) instance.removeLayer(id);

   319→        if (instance.getSource(id)) instance.removeSource(id);

   321→      dynamicLayersRef.current.clear();
   322→      dynamicSourcesRef.current.clear();

   342→    instance.setLayoutProperty('basemap-light-labels', 'visibility', showDark ? 'none' : 'visible');

   351→

   418→          fid,

   451→        {
   452→          id: 'gibs-layer',
   453→          type: 'raster',
   454→          source: 'gibs-satellite',
   455→          paint: { 'raster-opacity': 0.85 },
   456→        },
   457→        beforeId,
   458→      );
   459→      dynamicLayersRef.current.add('gibs-layer');
   460→    }
   461→    if (instance.getLayer('gibs-layer')) {
   462→      instance.setLayoutProperty('gibs-layer', 'visibility', layers.satellite ? 'visible' : 'none');
   463→    }
   464→  }, [mapReady, layers.satellite, radarData]);

   469→    if (!layers.quakes || quakes) return;

   477→        if (!controller.signal.aborted) setQuakes(data.quakes ?? []);

   480→        /* silent — the layer simply stays empty */

   483→  }, [layers.quakes, quakes]);
   484→
   485→  const quakesCollection = useMemo(
   486→    () => (quakes ? quakesToCollection(quakes) : null),
   487→    [quakes],
   488→  );
   489→
   490→  useEffect(() => {
   491→    const instance = mapRef.current;
   492→    if (!instance || !mapReady) return;
   493→    if (quakesCollection && !instance.getSource('quakes')) {
   494→      instance.addSource('quakes', { type: 'geojson', data: quakesCollection });
   495→      dynamicSourcesRef.current.add('quakes');
   496→      instance.addLayer(
   497→        {
   498→          id: 'quakes-circles',
   499→          type: 'circle',
   500→          source: 'quakes',
   501→          paint: {
   502→            'circle-radius': [
   503→              'interpolate',
   504→              ['linear'],
   505→              ['get', 'magnitude'],
   506→              2.5,
   507→              4,
   508→              4,
   509→              6,
   510→              5,
   511→              8.5,
   512→              7,
   513→              12,
   514→            ],
   515→            'circle-color': [
   516→              'case',
   517→              ['>=', ['get', 'magnitude'], 5],
   518→              '#E1552F',
   519→              ['>=', ['get', 'magnitude'], 4],
   520→              '#E8C54D',
   521→              '#45d1b3',
   522→            ],
   523→            'circle-opacity': 0.88,
   524→            'circle-stroke-width': 1.2,
   525→            'circle-stroke-color': 'rgba(255, 255, 255, 0.75)',
   526→          },
   527→        },
   528→        'overlay-anchor',
   529→      );
   530→      dynamicLayersRef.current.add('quakes-circles');
   531→    } else if (quakesCollection) {
   532→      (instance.getSource('quakes') as GeoJSONSource | undefined)?.setData(quakesCollection);
   533→    }
   534→    if (instance.getLayer('quakes-circles')) {
   535→      instance.setLayoutProperty('quakes-circles', 'visibility', layers.quakes ? 'visible' : 'none');
   536→    }
   537→  }, [mapReady, quakesCollection, layers.quakes]);
   538→
   539→  /* --------------------------- city pins ----------------------------- */
   540→
   541→  useEffect(() => {
   542→    const instance = mapRef.current;
   543→    if (!instance || !mapReady) return;
   544→    const markers: MLMarker[] = [];
   545→    const pins: Array<{ city: City; active: boolean }> = [
   546→      ...savedCities.filter((c) => c.id !== city?.id).map((c) => ({ city: c, active: false })),
   547→    ];
   548→    if (city) pins.push({ city, active: true });
   549→    for (const pin of pins) {
   550→      const el = document.createElement('button');
   551→      el.type = 'button';
   552→      el.className = `aether-pin focus-ring ${pin.active ? 'aether-pin--active' : ''}`;
   553→      el.title = pin.city.name;
   554→      el.setAttribute(
   555→        'aria-label',
   556→        pin.active ? `${pin.city.name} — current city` : pin.city.name,
   557→      );
   558→      el.addEventListener('click', () => {
   559→        instance.flyTo({
   560→          center: [pin.city.longitude, pin.city.latitude],
   561→          zoom: Math.max(instance.getZoom(), 7),
   562→          duration: 900,
   563→        });
   564→      });
   565→      markers.push(
   566→        new maplibregl.Marker({ element: el })
   567→          .setLngLat([pin.city.longitude, pin.city.latitude])
   568→          .addTo(instance),
   569→      );
   570→    }
   571→    return () => {
   572→      for (const m of markers) m.remove();
   573→    };
   574→  }, [mapReady, savedCities, city]);
   575→
   576→  /* ---------------------------- playback ----------------------------- */
   577→
   578→  useEffect(() => {
   579→    if (!radarPlaying || !layers.radar || radarTotal < 2) return;
   580→    const id = window.setInterval(() => {
   581→      if (document.hidden) return;
   582→      setRadarIndex((i) => (i + 1) % radarTotal);
   583→    }, 500);
   584→    return () => window.clearInterval(id);
   585→  }, [radarPlaying, layers.radar, radarTotal]);
   586→
   587→  useEffect(() => {
   588→    if (!hourPlaying || (!layers.precip && !layers.temp)) return;
   589→    const id = window.setInterval(() => {
   590→      if (document.hidden) return;
   591→      setHour((h) => (h + 1) % HOUR_STEPS);
   592→    }, 800);
   593→    return () => window.clearInterval(id);
   594→  }, [hourPlaying, layers.precip, layers.temp]);
   595→
   596→  /* ------------------------ keyboard + focus ------------------------- */
   597→
   598→  useEffect(() => {
   599→    const onKey = (event: KeyboardEvent) => {
   600→      if (event.key === 'Escape') {
   601→        if (popupDataRef.current) closePopup();
   602→        else onClose();
   603→        return;
   604→      }
   605→      if (event.key === 'Tab') {
   606→        const root = rootRef.current;
   607→        if (!root) return;
   608→        const focusables = Array.from(
   609→          root.querySelectorAll<HTMLElement>(
   610→            'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   611→          ),
   612→        ).filter((el) => (el as HTMLButtonElement).disabled !== true && el.offsetParent !== null);
   613→        if (focusables.length === 0) return;
   614→        const first = focusables[0];
   615→        const last = focusables[focusables.length - 1];
   616→        const active = document.activeElement;
   617→        if (event.shiftKey && (active === first || !root.contains(active))) {
   618→          event.preventDefault();
   619→          last.focus();
   620→        } else if (!event.shiftKey && (active === last || !root.contains(active))) {
   621→          event.preventDefault();
   622→          first.focus();
   623→        }
   624→      }
   625→    };
   626→    document.addEventListener('keydown', onKey);
   627→    return () => document.removeEventListener('keydown', onKey);
   628→  }, [closePopup, onClose]);
   629→
   630→  /* ----------------------------- chrome ------------------------------ */
   631→
   632→  const toggleLayer = useCallback((key: MapLayerKey) => {
   633→    setLayers((prev) => ({ ...prev, [key]: !prev[key] }));
   634→  }, []);
   635→
   636→  const handleBasemap = useCallback((kind: 'dark' | 'light') => {
   637→    setBasemapTouched(true);
   638→    setBasemap(kind);
   639→  }, []);
   640→
   641→  const radarTimeLabel = useMemo(() => {
   642→    if (radarTotal === 0) return radarError ? 'Radar unavailable' : null;
   643→    const frame = radarData!.frames[Math.min(radarIndex, radarTotal - 1)];
   644→    const time = new Date(frame.time * 1000).toLocaleTimeString([], {
   645→      hour: '2-digit',
   646→      minute: '2-digit',
   647→    });
   648→    const minutesAgo = Math.max(0, Math.round((Date.now() / 1000 - frame.time) / 60));
   649→    return minutesAgo < 1 ? `${time} · now` : `${time} · ${minutesAgo}m ago`;
   650→  }, [radarData, radarIndex, radarTotal, radarError]);
   651→
   652→  const legend: LegendInfo | null = useMemo(() => {
   653→    const note = hour === 0 ? 'Now' : `+${hour}h`;
   654→    if (layers.radar) {
   655→      return {
   656→        label: 'Radar',
   657→        ramp: RADAR_RAMP,
   658→        ticks: ['light', 'heavy'],
   659→        note: radarError ? 'Unavailable' : 'Past 2h · RainViewer',
   660→      };
   661→    }
   662→    if (layers.precip) {
   663→      return {
   664→        label: 'Precipitation',
   665→        ramp: PRECIP_RAMP,
   666→        ticks: ['0', '4', '8+ mm/h'],
   667→        note,
   668→      };
   669→    }
   670→    if (layers.temp) {
   671→      return {
   672→        label: 'Temperature',
   673→        ramp: TEMP_RAMP,
   674→        ticks: units === 'metric' ? ['-15°', '10°', '35°'] : ['5°', '50°', '95°'],
   675→        note,
   676→      };
   677→    }
   678→    if (layers.wind) {
   679→      return {
   680→        label: 'Wind',
   681→        ramp: WIND_RAMP,
   682→        ticks: units === 'metric' ? ['0', '30', '60+ km/h'] : ['0', '20', '40+ mph'],
   683→        note: windGrid.error ? 'Unavailable' : note,
   684→      };
   685→    }
   686→    return null;
   687→  }, [layers, hour, units, radarError, windGrid.error]);
   688→
   689→  const gridLoading = (layers.wind && windGrid.loading) || (layers.precip && precipGrid.loading) || (layers.temp && tempGrid.loading);
   690→
   691→  return (
   692→    <div
   693→      ref={rootRef}
   694→      role="dialog"
   695→      aria-modal="true"
   696→      aria-label="Map mode — live weather layers"
   697→      className="fixed inset-0 z-[80] flex flex-col"
   698→    >
   699→      {/* Top chrome — Clear LiquidGlass over the map (dimming scrim built in) */}
   700→      <LiquidGlass
   701→        variant="clear"
   702→        className="relative z-30 flex items-center gap-2 px-3 py-2"
   703→      >
   704→        <span
   705→          aria-hidden="true"
   706→          className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-aurora-300"
   707→        >
   708→          <Icon name="map" size={18} animated />
   709→        </span>
   710→        <div className="min-w-0 flex-1">
   711→          <p className="text-sm font-semibold leading-tight text-surface">Aether Map</p>
   712→          <p className="truncate text-[11px] leading-tight text-surface-muted">
   713→            {city
   714→              ? `${city.name}${city.country ? ` · ${city.country}` : ''}`
   715→              : 'Live weather layers around the world'}
   716→          </p>
   717→        </div>
   718→        <button
   719→          type="button"
   720→          onClick={() => setPickerOpen((open) => !open)}
   721→          aria-label={pickerOpen ? 'Close layers panel' : 'Open layers panel'}
   722→          aria-pressed={pickerOpen}
   723→          className={`focus-ring flex h-9 w-9 items-center justify-center rounded-full bg-white/10 transition-all duration-200 ${
   724→            pickerOpen
   725→              ? 'border-aurora-500/50 bg-aurora-500/15 text-aurora-400'
   726→              : 'text-surface-muted hover:text-surface'
   727→          }`}
   728→        >
   729→          <Icon name="layers" size={17} />
   730→        </button>
   731→        <button
   732→          ref={closeBtnRef}
   733→          type="button"
   734→          onClick={onClose}
   735→          aria-label="Close map (Escape)"
   736→          title="Close (Esc)"
   737→          className="focus-ring flex h-9 w-9 items-center justify-center rounded-full bg-white/10 text-white/85 transition-colors duration-200 hover:text-white"
   738→        >
   739→          <Icon name="x" size={17} />
   740→        </button>
   741→      </LiquidGlass>
   742→
   743→      {/* Map viewport */}
   744→      <div className="aether-map relative min-h-0 flex-1">
   745→        <div ref={containerRef} className="absolute inset-0" aria-label="Interactive weather map" />
   746→
   747→        {map && layers.precip && <HeatCanvas map={map} grid={precipGrid.data} metric="precip" />}
   748→        {map && layers.temp && <HeatCanvas map={map} grid={tempGrid.data} metric="temp" />}
   749→        {map && layers.wind && <WindParticles map={map} grid={windGrid.data} />}
   750→
   751→        <MapLegend legend={legend} loading={gridLoading} />
   752→
   753→        <TimeScrubber
   754→          showRadar={layers.radar}
   755→          radarIndex={radarIndex}
   756→          radarTotal={radarTotal}
   757→          radarTimeLabel={radarTimeLabel}
   758→          radarHiddenByZoom={radarHiddenByZoom}
   759→          radarPlaying={radarPlaying}
   760→          onRadarIndex={setRadarIndex}
   761→          onRadarPlaying={setRadarPlaying}
   762→          showHour={layers.precip || layers.temp}
   763→          hour={hour}
   764→          hourPlaying={hourPlaying}
   765→          onHour={setHour}
   766→          onHourPlaying={setHourPlaying}
   767→        />
   768→
   769→        <LayerPicker
   770→          open={pickerOpen}
   771→          onClose={() => setPickerOpen(false)}
   772→          layers={layers}
   773→          onToggleLayer={toggleLayer}
   774→          basemap={basemap}
   775→          onBasemap={handleBasemap}
   776→        />
   777→
   778→        {popupContainer &&
   779→          popupData &&
   780→          createPortal(
   781→            popupData.kind === 'point' ? (
   782→              <PointPopup
   783→                key={`${popupData.lng},${popupData.lat}`}
   784→                lng={popupData.lng}
   785→                lat={popupData.lat}
   786→                units={units}
   787→                onClose={closePopup}
   788→              />
   789→            ) : (
   790→              <QuakePopup key={popupData.quake.id} quake={popupData.quake} onClose={closePopup} />
   791→            ),
   792→            popupContainer,
   793→          )}
   794→      </div>
   795→    </div>
   796→  );
   797→}
