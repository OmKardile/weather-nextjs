    71→
    72→function normalizeQuake(p: Record<string, unknown>): GeoQuake {
    73→  return {
    74→    id: String(p.id ?? ''),
    75→    magnitude: Number(p.magnitude ?? 0),
    76→    place: String(p.place ?? ''),
    77→    time: String(p.time ?? new Date().toISOString()),
    78→    latitude: Number(p.latitude ?? 0),
    79→    longitude: Number(p.longitude ?? 0),
    80→    depthKm: Number(p.depthKm ?? 0),
    81→    url: String(p.url ?? ''),
    82→    tsunami: Boolean(p.tsunami),
    83→  };
    84→}
    85→
    86→function quakesToCollection(quakes: GeoQuake[]): FeatureCollection {
    87→  return {
    88→    type: 'FeatureCollection',
    89→    features: quakes.map((q) => ({
    90→      type: 'Feature' as const,
    91→      geometry: { type: 'Point' as const, coordinates: [q.longitude, q.latitude] },
    92→      properties: { ...q },
    93→    })),
    94→  };
    95→}

   122→  const [layers, setLayers] = useState<LayerState>({

   140→  const [radarHiddenByZoom, setRadarHiddenByZoom] = useState(false);

   157→  }, []);
   158→

   160→

   176→      const host = document.createElement('div');

   216→
   217→  /* --------------------------- map init ------------------------------ */

   219→  useEffect(() => {

   222→

   224→    const instance = new maplibregl.Map({

   228→      zoom: initial ? 5.2 : 1.9,
   229→      minZoom: 1,
   230→      maxZoom: 17,
   231→      dragRotate: false,
   232→      touchPitch: false,
   233→      attributionControl: { compact: true },
   234→    });
   235→    mapRef.current = instance;
   236→    instance.addControl(new maplibregl.NavigationControl({ showCompass: false }), 'top-left');
   237→    // Radar/satellite tiles 404 at the edges — expected, stay quiet.
   238→    instance.on('error', () => {});
   239→
   240→    const round1 = (v: number) => Math.round(v * 10) / 10;
   241→    const syncBbox = () => {
   242→      const b = instance.getBounds();
   243→      const next: ViewBbox = {
   244→        minLat: round1(b.getSouth()),
   245→        minLon: round1(b.getWest()),

   252→        prev.minLat !== next.minLat ||
   253→        prev.minLon !== next.minLon ||
   254→        prev.maxLat !== next.maxLat ||
   255→        prev.maxLon !== next.maxLon

   269→      setMap(instance);
   270→      syncBbox();
   271→    };
   272→    instance.on('load', onLoad);
   273→    instance.on('moveend', onMoveEnd);
   274→
   275→    // RainViewer free tier: radar hides above zoom 7.

   280→        setRadarHiddenByZoom(hide);

   283→    instance.on('zoom', onZoom);
   284→
   285→    const onClick = (e: maplibregl.MapMouseEvent) => {

   290→          return;
   291→        }
   292→      }
   293→      openPointPopup(e.lngLat.lng, e.lngLat.lat);
   294→    };
   295→    instance.on('click', onClick);
   296→
   297→    const onMouseMove = (e: maplibregl.MapMouseEvent) => {
   298→      if (!layersRef.current?.quakes || !instance.getLayer('quakes-circles')) return;

   300→      instance.getCanvas().style.cursor = hits.length > 0 ? 'pointer' : '';
   301→    };
   302→    instance.on('mousemove', onMouseMove);
   303→
   304→    return () => {
   305→      window.clearTimeout(bboxDebounce);
   306→      popupRef.current?.remove();
   307→      markerRef.current?.remove();
   308→      popupRef.current = null;
   309→      markerRef.current = null;
   310→      // Explicit removal of every dynamic layer/source, then the map itself.
   311→      for (const id of dynamicLayersRef.current) {
   312→        if (instance.getLayer(id)) instance.removeLayer(id);
   313→      }
   314→      for (const id of dynamicSourcesRef.current) {
   315→        if (instance.getSource(id)) instance.removeSource(id);
   316→      }
   317→      dynamicLayersRef.current.clear();
   318→      dynamicSourcesRef.current.clear();

   333→    if (!instance || !mapReady) return;
   334→    const showDark = basemap === 'dark';

   336→    instance.setLayoutProperty('basemap-dark-labels', 'visibility', showDark ? 'visible' : 'none');
   337→    instance.setLayoutProperty('basemap-light', 'visibility', showDark ? 'none' : 'visible');
   338→    instance.setLayoutProperty('basemap-light-labels', 'visibility', showDark ? 'none' : 'visible');
   339→    instance.setPaintProperty('bg', 'background-color', BASEMAP_BG[basemap]);
   340→  }, [mapReady, basemap]);

   344→  const windGrid = useGrid('wind', viewBbox, hour, layers.wind);

   347→

   402→            type: 'raster',

   407→        );
   408→        dynamicLayersRef.current.add(id);

   414→          fid,
   415→          'visibility',
   416→          frameItem === frame && !radarHiddenByZoom ? 'visible' : 'none',

   445→        radarLayer ?? (instance.getLayer('quakes-circles') ? 'quakes-circles' : 'overlay-anchor');

   447→        {
   448→          id: 'gibs-layer',
   449→          type: 'raster',
   450→          source: 'gibs-satellite',
   451→          paint: { 'raster-opacity': 0.85 },
   452→        },
   453→        beforeId,
   454→      );
   455→      dynamicLayersRef.current.add('gibs-layer');
   456→    }
   457→    if (instance.getLayer('gibs-layer')) {
   458→      instance.setLayoutProperty('gibs-layer', 'visibility', layers.satellite ? 'visible' : 'none');
   459→    }
   460→  }, [mapReady, layers.satellite, radarData]);

   462→  /* --------------------------- earthquakes --------------------------- */
   463→
   464→  useEffect(() => {
   465→    if (!layers.quakes || quakes) return;

   469→        if (!res.ok) throw new Error(`Earthquake feed failed (${res.status})`);

   473→        if (!controller.signal.aborted) setQuakes(data.quakes ?? []);

   475→      .catch(() => {
   476→        /* silent — the layer simply stays empty */

   479→  }, [layers.quakes, quakes]);
   480→
   481→  const quakesCollection = useMemo(
   482→    () => (quakes ? quakesToCollection(quakes) : null),
   483→    [quakes],
   484→  );
   485→
   486→  useEffect(() => {
   487→    const instance = mapRef.current;
   488→    if (!instance || !mapReady) return;
   489→    if (quakesCollection && !instance.getSource('quakes')) {
   490→      instance.addSource('quakes', { type: 'geojson', data: quakesCollection });
   491→      dynamicSourcesRef.current.add('quakes');
   492→      instance.addLayer(
   493→        {
   494→          id: 'quakes-circles',
   495→          type: 'circle',
   496→          source: 'quakes',
   497→          paint: {
   498→            'circle-radius': [
   499→              'interpolate',
   500→              ['linear'],
   501→              ['get', 'magnitude'],
   502→              2.5,
   503→              4,
   504→              4,
   505→              6,
   506→              5,
   507→              8.5,
   508→              7,
   509→              12,
   510→            ],
   511→            'circle-color': [
   512→              'case',
   513→              ['>=', ['get', 'magnitude'], 5],
   514→              '#E1552F',
   515→              ['>=', ['get', 'magnitude'], 4],
   516→              '#E8C54D',
   517→              '#45d1b3',
   518→            ],
   519→            'circle-opacity': 0.88,
   520→            'circle-stroke-width': 1.2,
   521→            'circle-stroke-color': 'rgba(255, 255, 255, 0.75)',
   522→          },
   523→        },
   524→        'overlay-anchor',
   525→      );
   526→      dynamicLayersRef.current.add('quakes-circles');
   527→    } else if (quakesCollection) {
   528→      (instance.getSource('quakes') as GeoJSONSource | undefined)?.setData(quakesCollection);
   529→    }
   530→    if (instance.getLayer('quakes-circles')) {
   531→      instance.setLayoutProperty('quakes-circles', 'visibility', layers.quakes ? 'visible' : 'none');
   532→    }
   533→  }, [mapReady, quakesCollection, layers.quakes]);
   534→
   535→  /* --------------------------- city pins ----------------------------- */
   536→
   537→  useEffect(() => {
   538→    const instance = mapRef.current;
   539→    if (!instance || !mapReady) return;
   540→    const markers: MLMarker[] = [];
   541→    const pins: Array<{ city: City; active: boolean }> = [
   542→      ...savedCities.filter((c) => c.id !== city?.id).map((c) => ({ city: c, active: false })),
   543→    ];
   544→    if (city) pins.push({ city, active: true });
   545→    for (const pin of pins) {
   546→      const el = document.createElement('button');
   547→      el.type = 'button';
   548→      el.className = `aether-pin focus-ring ${pin.active ? 'aether-pin--active' : ''}`;
   549→      el.title = pin.city.name;
   550→      el.setAttribute(
   551→        'aria-label',
   552→        pin.active ? `${pin.city.name} — current city` : pin.city.name,
   553→      );
   554→      el.addEventListener('click', () => {
   555→        instance.flyTo({
   556→          center: [pin.city.longitude, pin.city.latitude],
   557→          zoom: Math.max(instance.getZoom(), 7),
   558→          duration: 900,
   559→        });
   560→      });
   561→      markers.push(
   562→        new maplibregl.Marker({ element: el })
   563→          .setLngLat([pin.city.longitude, pin.city.latitude])
   564→          .addTo(instance),
   565→      );
   566→    }
   567→    return () => {
   568→      for (const m of markers) m.remove();
   569→    };
   570→  }, [mapReady, savedCities, city]);
   571→
   572→  /* ---------------------------- playback ----------------------------- */
   573→
   574→  useEffect(() => {
   575→    if (!radarPlaying || !layers.radar || radarTotal < 2) return;
   576→    const id = window.setInterval(() => {
   577→      if (document.hidden) return;
   578→      setRadarIndex((i) => (i + 1) % radarTotal);
   579→    }, 500);
   580→    return () => window.clearInterval(id);
   581→  }, [radarPlaying, layers.radar, radarTotal]);
   582→
   583→  useEffect(() => {
   584→    if (!hourPlaying || (!layers.precip && !layers.temp)) return;
   585→    const id = window.setInterval(() => {
   586→      if (document.hidden) return;
   587→      setHour((h) => (h + 1) % HOUR_STEPS);
   588→    }, 800);
   589→    return () => window.clearInterval(id);
   590→  }, [hourPlaying, layers.precip, layers.temp]);
   591→
   592→  /* ------------------------ keyboard + focus ------------------------- */
   593→
   594→  useEffect(() => {
   595→    const onKey = (event: KeyboardEvent) => {
   596→      if (event.key === 'Escape') {
   597→        if (popupDataRef.current) closePopup();
   598→        else onClose();
   599→        return;
   600→      }
   601→      if (event.key === 'Tab') {
   602→        const root = rootRef.current;
   603→        if (!root) return;
   604→        const focusables = Array.from(
   605→          root.querySelectorAll<HTMLElement>(
   606→            'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   607→          ),
   608→        ).filter((el) => (el as HTMLButtonElement).disabled !== true && el.offsetParent !== null);
   609→        if (focusables.length === 0) return;
   610→        const first = focusables[0];
   611→        const last = focusables[focusables.length - 1];
   612→        const active = document.activeElement;
   613→        if (event.shiftKey && (active === first || !root.contains(active))) {
   614→          event.preventDefault();
   615→          last.focus();
   616→        } else if (!event.shiftKey && (active === last || !root.contains(active))) {
   617→          event.preventDefault();
   618→          first.focus();
   619→        }
   620→      }
   621→    };
   622→    document.addEventListener('keydown', onKey);
   623→    return () => document.removeEventListener('keydown', onKey);
   624→  }, [closePopup, onClose]);
   625→
   626→  /* ----------------------------- chrome ------------------------------ */
   627→
   628→  const toggleLayer = useCallback((key: MapLayerKey) => {
   629→    setLayers((prev) => ({ ...prev, [key]: !prev[key] }));
   630→  }, []);
   631→
   632→  const handleBasemap = useCallback((kind: 'dark' | 'light') => {
   633→    setBasemapTouched(true);
   634→    setBasemap(kind);
   635→  }, []);
   636→
   637→  const radarTimeLabel = useMemo(() => {
   638→    if (radarTotal === 0) return radarError ? 'Radar unavailable' : null;
   639→    const frame = radarData!.frames[Math.min(radarIndex, radarTotal - 1)];
   640→    const time = new Date(frame.time * 1000).toLocaleTimeString([], {
   641→      hour: '2-digit',
   642→      minute: '2-digit',
   643→    });
   644→    const minutesAgo = Math.max(0, Math.round((Date.now() / 1000 - frame.time) / 60));
   645→    return minutesAgo < 1 ? `${time} · now` : `${time} · ${minutesAgo}m ago`;
   646→  }, [radarData, radarIndex, radarTotal, radarError]);
   647→
   648→  const legend: LegendInfo | null = useMemo(() => {
   649→    const note = hour === 0 ? 'Now' : `+${hour}h`;
   650→    if (layers.radar) {
   651→      return {
   652→        label: 'Radar',
   653→        ramp: RADAR_RAMP,
   654→        ticks: ['light', 'heavy'],
   655→        note: radarError ? 'Unavailable' : 'Past 2h · RainViewer',
   656→      };
   657→    }
   658→    if (layers.precip) {
   659→      return {
   660→        label: 'Precipitation',
   661→        ramp: PRECIP_RAMP,
   662→        ticks: ['0', '4', '8+ mm/h'],
   663→        note,
   664→      };
   665→    }
   666→    if (layers.temp) {
   667→      return {
   668→        label: 'Temperature',
   669→        ramp: TEMP_RAMP,
   670→        ticks: units === 'metric' ? ['-15°', '10°', '35°'] : ['5°', '50°', '95°'],
   671→        note,
   672→      };
   673→    }
   674→    if (layers.wind) {
   675→      return {
   676→        label: 'Wind',
   677→        ramp: WIND_RAMP,
   678→        ticks: units === 'metric' ? ['0', '30', '60+ km/h'] : ['0', '20', '40+ mph'],
   679→        note: windGrid.error ? 'Unavailable' : note,
   680→      };
   681→    }
   682→    return null;
   683→  }, [layers, hour, units, radarError, windGrid.error]);
   684→
   685→  const gridLoading = (layers.wind && windGrid.loading) || (layers.precip && precipGrid.loading) || (layers.temp && tempGrid.loading);
   686→
   687→  return (
   688→    <div
   689→      ref={rootRef}
   690→      role="dialog"
   691→      aria-modal="true"
   692→      aria-label="Map mode — live weather layers"
   693→      className="fixed inset-0 z-[80] flex flex-col"
   694→    >
   695→      {/* Top chrome */}
   696→      <Glass level="overlay" className="relative z-30 flex items-center gap-2 rounded-none px-3 py-2">
   697→        <span
   698→          aria-hidden="true"
   699→          className="glass-subtle flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-aurora-400"
   700→        >
   701→          <Icon name="map" size={18} animated />
   702→        </span>
   703→        <div className="min-w-0 flex-1">
   704→          <p className="text-sm font-semibold leading-tight text-surface">Aether Map</p>
   705→          <p className="truncate text-[11px] leading-tight text-surface-muted">
   706→            {city
   707→              ? `${city.name}${city.country ? ` · ${city.country}` : ''}`
   708→              : 'Live weather layers around the world'}
   709→          </p>
   710→        </div>
   711→        <button
   712→          type="button"
   713→          onClick={() => setPickerOpen((open) => !open)}
   714→          aria-label={pickerOpen ? 'Close layers panel' : 'Open layers panel'}
   715→          aria-pressed={pickerOpen}
   716→          className={`glass-subtle focus-ring flex h-9 w-9 items-center justify-center rounded-full transition-all duration-200 ${
   717→            pickerOpen
   718→              ? 'border-aurora-500/50 bg-aurora-500/15 text-aurora-400'
   719→              : 'text-surface-muted hover:text-surface'
   720→          }`}
   721→        >
   722→          <Icon name="layers" size={17} />
   723→        </button>
   724→        <button
   725→          ref={closeBtnRef}
   726→          type="button"
   727→          onClick={onClose}
   728→          aria-label="Close map (Escape)"
   729→          title="Close (Esc)"
   730→          className="glass-subtle focus-ring flex h-9 w-9 items-center justify-center rounded-full text-surface-muted transition-all duration-200 hover:text-surface"
   731→        >
   732→          <Icon name="x" size={17} />
   733→        </button>
   734→      </Glass>
   735→
   736→      {/* Map viewport */}
   737→      <div className="aether-map relative min-h-0 flex-1">
   738→        <div ref={containerRef} className="absolute inset-0" aria-label="Interactive weather map" />
   739→
   740→        {map && layers.precip && <HeatCanvas map={map} grid={precipGrid.data} metric="precip" />}
   741→        {map && layers.temp && <HeatCanvas map={map} grid={tempGrid.data} metric="temp" />}
   742→        {map && layers.wind && <WindParticles map={map} grid={windGrid.data} />}
   743→
   744→        <MapLegend legend={legend} loading={gridLoading} />
   745→
   746→        <TimeScrubber
   747→          showRadar={layers.radar}
   748→          radarIndex={radarIndex}
   749→          radarTotal={radarTotal}
   750→          radarTimeLabel={radarTimeLabel}
   751→          radarHiddenByZoom={radarHiddenByZoom}
   752→          radarPlaying={radarPlaying}
   753→          onRadarIndex={setRadarIndex}
   754→          onRadarPlaying={setRadarPlaying}
   755→          showHour={layers.precip || layers.temp}
   756→          hour={hour}
   757→          hourPlaying={hourPlaying}
   758→          onHour={setHour}
   759→          onHourPlaying={setHourPlaying}
   760→        />
   761→
   762→        <LayerPicker
   763→          open={pickerOpen}
   764→          onClose={() => setPickerOpen(false)}
   765→          layers={layers}
   766→          onToggleLayer={toggleLayer}
   767→          basemap={basemap}
   768→          onBasemap={handleBasemap}
   769→        />
   770→
   771→        {popupContainer &&
   772→          popupData &&
   773→          createPortal(
   774→            popupData.kind === 'point' ? (
   775→              <PointPopup
   776→                key={`${popupData.lng},${popupData.lat}`}
   777→                lng={popupData.lng}
   778→                lat={popupData.lat}
   779→                units={units}
   780→                onClose={closePopup}
   781→              />
   782→            ) : (
   783→              <QuakePopup key={popupData.quake.id} quake={popupData.quake} onClose={closePopup} />
   784→            ),
   785→            popupContainer,
   786→          )}
   787→      </div>
   788→    </div>
   789→  );
   790→}
